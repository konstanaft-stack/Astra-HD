cmake_minimum_required(VERSION 3.24)
project(EditorSceneImageTests LANGUAGES CXX)
enable_testing()
set(OPENXR_INCLUDE_DIR "" CACHE PATH "OpenXR include directory")
find_path(OPENXR_HEADERS openxr/openxr.h HINTS "${OPENXR_INCLUDE_DIR}" REQUIRED)
add_executable(editor_scene_image_tests main.cpp ../../src/MenuPreview/EditorSceneImage.cpp ../../src/Render/NativeColorSnapshots.cpp ../../src/Render/CommandResources.cpp)
target_include_directories(editor_scene_image_tests PRIVATE ../../include "${OPENXR_HEADERS}")
target_compile_features(editor_scene_image_tests PRIVATE cxx_std_20)
target_compile_definitions(editor_scene_image_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
 set_property(TARGET editor_scene_image_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
target_link_libraries(editor_scene_image_tests PRIVATE d3d12 dxgi)
add_test(NAME editor_scene_image_warp COMMAND editor_scene_image_tests)
