cmake_minimum_required(VERSION 3.20)
project(editor_freshness_tests LANGUAGES CXX)
add_executable(editor_freshness_tests main.cpp)
target_compile_features(editor_freshness_tests PRIVATE cxx_std_20)
find_package(Threads REQUIRED)
target_link_libraries(editor_freshness_tests PRIVATE Threads::Threads)
target_include_directories(editor_freshness_tests PRIVATE ../../include)
enable_testing()
add_test(NAME editor_freshness COMMAND editor_freshness_tests)
