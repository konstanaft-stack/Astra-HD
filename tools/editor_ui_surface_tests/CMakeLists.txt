cmake_minimum_required(VERSION 3.24)
project(EditorUiSurfaceTests LANGUAGES CXX)
enable_testing()
set(OPENXR_INCLUDE_DIR "" CACHE PATH "Directory containing openxr/openxr.h")
find_path(OPENXR_HEADERS openxr/openxr.h HINTS "${OPENXR_INCLUDE_DIR}" REQUIRED)
add_executable(editor_ui_surface_tests main.cpp ../../src/Render/CommandResources.cpp)
# Allow a linker RED before the implementation exists, using the same harness.
option(EDITOR_UI_SURFACE_LINK_IMPLEMENTATION "Link the production EditorUiSurface implementation" ON)
if(EDITOR_UI_SURFACE_LINK_IMPLEMENTATION AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/MenuPreview/EditorUiSurface.cpp")
  target_sources(editor_ui_surface_tests PRIVATE ../../src/MenuPreview/EditorUiSurface.cpp)
endif()
target_include_directories(editor_ui_surface_tests PRIVATE ../../include "${OPENXR_HEADERS}")
target_compile_features(editor_ui_surface_tests PRIVATE cxx_std_20)
if(MSVC)
  set_property(TARGET editor_ui_surface_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
target_compile_definitions(editor_ui_surface_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(editor_ui_surface_tests PRIVATE d3d12 dxgi)
add_test(NAME editor_ui_surface_warp COMMAND editor_ui_surface_tests)
# Separate processes exercise the real startup-cached production ENV parser.
add_test(NAME editor_ui_activation_config0 COMMAND editor_ui_surface_tests --activation 0 unset 0)
add_test(NAME editor_ui_activation_config1 COMMAND editor_ui_surface_tests --activation 1 unset 1)
add_test(NAME editor_ui_activation_force0 COMMAND editor_ui_surface_tests --activation 1 0 0)
add_test(NAME editor_ui_activation_force1 COMMAND editor_ui_surface_tests --activation 0 1 1)
add_test(NAME editor_ui_activation_malformed0 COMMAND editor_ui_surface_tests --activation 0 01 0)
add_test(NAME editor_ui_activation_malformed1 COMMAND editor_ui_surface_tests --activation 1 malformed 1)
