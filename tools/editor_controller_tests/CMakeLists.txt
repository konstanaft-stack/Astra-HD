cmake_minimum_required(VERSION 3.24)
project(EditorControllerTests LANGUAGES CXX)
enable_testing()
set(EDITOR_CONTROLLER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/MenuPreview/EditorControllerPolicy.cpp" CACHE FILEPATH "Production policy source")
add_executable(editor_controller_tests main.cpp "${EDITOR_CONTROLLER_SOURCE}")
target_include_directories(editor_controller_tests PRIVATE ../../include)
target_compile_features(editor_controller_tests PRIVATE cxx_std_20)
target_compile_definitions(editor_controller_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
 set_property(TARGET editor_controller_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
add_test(NAME editor_controller_policy COMMAND editor_controller_tests)
